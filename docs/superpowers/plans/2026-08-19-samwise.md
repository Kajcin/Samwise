# Samwise Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Samwise end-to-end — a Supabase-backed Angular app for planning garden bed layouts, tracking plants, auto-generating care tasks, and syncing them to Google Calendar — following the five-stage build order from the design spec.

**Architecture:** An Angular standalone-component frontend talks directly to Supabase (Postgres + Row Level Security + Auth) via `@supabase/supabase-js`. All task-generation logic lives in a Postgres trigger/function, never the client. A single Supabase Edge Function handles the Google Calendar push and is invoked by a DB webhook on `tasks` writes, keeping the Calendar refresh token server-side only.

**Tech Stack:** Angular (standalone components, Router, signals), TypeScript, Supabase (Postgres, Row Level Security, Auth, Edge Functions, pg_cron), Jasmine/Karma via the Angular CLI (pinned with `--test-runner=karma`), Deno + `Deno.test` for Edge Functions, Supabase CLI as a dev dependency (`npx supabase …`), `psql` via `npm run db:psql` (local psql or the one in the DB container, always `ON_ERROR_STOP=1`) for SQL-level tests against a local Supabase instance (`supabase start`), GitHub Actions CI running all three layers.

**Spec:** `docs/superpowers/specs/2026-08-19-samwise-design.md`

## Global Constraints

- Single user for now; every table except `plants_catalog`'s seeded rows carries `user_id` with RLS `user_id = auth.uid()`, so multi-user later needs no schema rework.
- Responsive web app (desktop + mobile browser) — no native app, no offline-first PWA.
- Garden bed layout is a fixed rows × cols grid — no free-form/canvas layouts.
- No third-party plant API — plant data is seeded once, plus user-entered custom plants.
- Google is the sole OAuth provider. Sign-in requests only basic profile scope; the `calendar.app.created` scope is requested separately, only when the user clicks "Connect Calendar".
- Task generation (`seed_start` / `transplant` / `prune` / `fertilize` / `harvest`) runs in a Postgres function + trigger, never client code.
- Events go to a dedicated "Samwise" secondary calendar that the app creates, never to the user's primary calendar.
- The Google Calendar refresh token is stored and used server-side only (Edge Functions), never sent to the client. The client has no access to `calendar_credentials`; it calls the `has_calendar_connection()` RPC instead.
- The `tasks` → `calendar-sync` webhook authenticates with a shared secret (`x-webhook-secret` header). The function URL and the secret live in Supabase Vault.
- Edge Function logic that tests need lives outside `index.ts`, so importing it in a test doesn't start `Deno.serve`.

`supabase/migrations/0001_core_schema.sql`:
```sql
create extension if not exists pgcrypto;

create table plants_catalog (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references auth.users(id),
  name text not null,
  species text,
  sun_requirements text check (sun_requirements in ('full_sun','partial_shade','full_shade')),
  watering_frequency_days int,
  flowering_start_month int check (flowering_start_month between 1 and 12),
  flowering_end_month int check (flowering_end_month between 1 and 12),
  notes text
);

create table care_rules (
  id uuid primary key default gen_random_uuid(),
  plant_id uuid not null references plants_catalog(id) on delete cascade,
  task_type text not null check (task_type in ('seed_start','transplant','prune','fertilize','harvest')),
  timing_type text not null check (timing_type in ('offset_from_planting','recurring')),
  offset_days int,
  recurrence_days int,
  notes text,
  constraint offset_has_days check (timing_type <> 'offset_from_planting' or offset_days is not null),
  constraint recurring_has_days check (timing_type <> 'recurring' or recurrence_days is not null)
);

create table beds (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id),
  name text not null,
  rows int not null check (rows > 0),
  cols int not null check (cols > 0)
);

create table plantings (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id),
  bed_id uuid not null references beds(id) on delete cascade,
  cell_row int not null,
  cell_col int not null,
  plant_id uuid not null references plants_catalog(id),
  planted_date date not null,
  status text not null default 'active' check (status in ('active','removed'))
);

create unique index plantings_one_active_per_cell
  on plantings (bed_id, cell_row, cell_col)
  where status = 'active';

create table tasks (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id),
  planting_id uuid references plantings(id) on delete cascade,
  task_type text not null check (task_type in ('seed_start','transplant','prune','fertilize','harvest')),
  due_date date not null,
  completed_at timestamptz,
  google_calendar_event_id text
);

create table calendar_credentials (
  user_id uuid primary key references auth.users(id),
  refresh_token text not null,
  access_token text,
  expires_at timestamptz,
  calendar_id text not null
);

alter table plants_catalog enable row level security;
alter table care_rules enable row level security;
alter table beds enable row level security;
alter table plantings enable row level security;
alter table tasks enable row level security;
alter table calendar_credentials enable row level security;

create policy plants_catalog_select on plants_catalog
  for select using (user_id is null or user_id = auth.uid());
create policy plants_catalog_write on plants_catalog
  for all using (user_id = auth.uid()) with check (user_id = auth.uid());

create policy care_rules_select on care_rules
  for select using (
    exists (select 1 from plants_catalog p where p.id = plant_id and (p.user_id is null or p.user_id = auth.uid()))
  );
create policy care_rules_write on care_rules
  for all using (
    exists (select 1 from plants_catalog p where p.id = plant_id and p.user_id = auth.uid())
  ) with check (
    exists (select 1 from plants_catalog p where p.id = plant_id and p.user_id = auth.uid())
  );

create policy beds_all on beds
  for all using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy plantings_all on plantings
  for all using (user_id = auth.uid()) with check (user_id = auth.uid());
create policy tasks_all on tasks
  for all using (user_id = auth.uid()) with check (user_id = auth.uid());

-- calendar_credentials holds the Google refresh token: no client access at all.
-- RLS is enabled with no policies (deny-all), and table grants are revoked as a
-- second layer. Edge Functions use the service role, which bypasses RLS.
revoke all on calendar_credentials from anon, authenticated;
```

`src/app/core/auth.service.ts`:
```typescript
import { Injectable, signal } from '@angular/core';
import type { User } from '@supabase/supabase-js';
import { supabase } from './supabase-client';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private currentUserSignal = signal<User | null>(null);
  currentUser = this.currentUserSignal.asReadonly();

  constructor() {
    supabase.auth.getSession().then(({ data }) => {
      this.currentUserSignal.set(data.session?.user ?? null);
    });
    supabase.auth.onAuthStateChange((_event, session) => {
      this.currentUserSignal.set(session?.user ?? null);
    });
  }

  async signInWithGoogle(): Promise<void> {
    await supabase.auth.signInWithOAuth({
      provider: 'google',
      options: { scopes: 'email profile' },
    });
  }

  async signOut(): Promise<void> {
    await supabase.auth.signOut();
    this.currentUserSignal.set(null);
  }
}
```

```javascript
#!/usr/bin/env node
// Resets the local Supabase DB (migrations + seed) and runs every SQL test file.
// Requires a running local stack (`npx supabase start`). Stops on the first failure.
import { spawnSync } from 'node:child_process';
import { readdirSync } from 'node:fs';
import { runPsql } from './psql.mjs';

const dir = 'supabase/tests/sql';

const reset = spawnSync('npx --no-install supabase db reset', { stdio: 'inherit', shell: true });
if (reset.status !== 0) process.exit(reset.status ?? 1);

let files = [];
try {
  files = readdirSync(dir).filter((f) => f.endsWith('.test.sql')).sort();
} catch {
  // no tests directory yet
}
if (files.length === 0) {
  console.log('No SQL tests found.');
  process.exit(0);
}

for (const file of files) {
  console.log(`== ${dir}/${file}`);
  const status = runPsql(['-q', '-f', `${dir}/${file}`]);
  if (status !== 0) process.exit(status);
}
console.log(`All ${files.length} SQL test files passed.`);
```

```javascript
#!/usr/bin/env node
// Runs Deno unit tests and type-checks every Edge Function entry point.
// A no-op until Stage 5 adds supabase/functions.
import { spawnSync } from 'node:child_process';
import { existsSync, readdirSync } from 'node:fs';
import { join } from 'node:path';

const root = 'supabase/functions';
if (!existsSync(root)) {
  console.log('No Edge Functions yet.');
  process.exit(0);
}

const files = readdirSync(root, { recursive: true }).map((p) => join(root, String(p)).replaceAll('\\', '/'));

function deno(args) {
  const result = spawnSync('deno', args, { stdio: 'inherit' });
  if (result.error) {
    console.error(`Failed to start deno (is it installed?): ${result.error.message}`);
    process.exit(1);
  }
  if (result.status !== 0) process.exit(result.status ?? 1);
}

if (files.some((f) => f.endsWith('.test.ts'))) deno(['test', root]);

// index.ts files aren't unit-tested (they only wire requests to tested modules),
// so at least make sure they type-check.
const entryPoints = files.filter((f) => f.endsWith('/index.ts'));
if (entryPoints.length) deno(['check', ...entryPoints]);
```

```typescript
import { Injectable } from '@angular/core';
import { supabase } from '../core/supabase-client';

export interface Plant {
  id: string;
  user_id: string | null;
  name: string;
  species: string | null;
  sun_requirements: 'full_sun' | 'partial_shade' | 'full_shade' | null;
  watering_frequency_days: number | null;
  flowering_start_month: number | null;
  flowering_end_month: number | null;
  notes: string | null;
}

export type NewPlant = Omit<Plant, 'id' | 'user_id'>;

export interface CareRule {
  id: string;
  plant_id: string;
  task_type: 'seed_start' | 'transplant' | 'prune' | 'fertilize' | 'harvest';
  timing_type: 'offset_from_planting' | 'recurring';
  offset_days: number | null;
  recurrence_days: number | null;
  notes: string | null;
}

export type NewCareRule = Omit<CareRule, 'id' | 'plant_id'>;

@Injectable({ providedIn: 'root' })
export class PlantCatalogService {
  async list(): Promise<Plant[]> {
    const { data, error } = await supabase.from('plants_catalog').select('*').order('name');
    if (error) throw error;
    return data as Plant[];
  }

  async addCustom(plant: NewPlant): Promise<Plant> {
    const { data: userData } = await supabase.auth.getUser();
    const { data, error } = await supabase
      .from('plants_catalog')
      .insert({ ...plant, user_id: userData.user!.id })
      .select()
      .single();
    if (error) throw error;
    return data as Plant;
  }

  async cloneToCustom(plantId: string): Promise<Plant> {
    const { data: source, error: sourceError } = await supabase
      .from('plants_catalog')
      .select('*')
      .eq('id', plantId)
      .single();
    if (sourceError) throw sourceError;

    const { data: rules, error: rulesError } = await supabase
      .from('care_rules')
      .select('task_type,timing_type,offset_days,recurrence_days,notes')
      .eq('plant_id', plantId);
    if (rulesError) throw rulesError;

    const { data: userData } = await supabase.auth.getUser();
    const { id, user_id, ...fields } = source as Plant;
    const { data: clone, error: cloneError } = await supabase
      .from('plants_catalog')
      .insert({ ...fields, user_id: userData.user!.id })
      .select()
      .single();
    if (cloneError) throw cloneError;

    if (rules && rules.length) {
      const { error: insertRulesError } = await supabase
        .from('care_rules')
        .insert(rules.map(r => ({ ...r, plant_id: (clone as Plant).id })));
      if (insertRulesError) throw insertRulesError;
    }

    return clone as Plant;
  }

  async listCareRules(plantId: string): Promise<CareRule[]> {
    const { data, error } = await supabase.from('care_rules').select('*').eq('plant_id', plantId);
    if (error) throw error;
    return data as CareRule[];
  }

  async addCareRule(plantId: string, rule: NewCareRule): Promise<CareRule> {
    const { data, error } = await supabase
      .from('care_rules')
      .insert({ ...rule, plant_id: plantId })
      .select()
      .single();
    if (error) throw error;
    return data as CareRule;
  }

  async updateCareRule(ruleId: string, changes: Partial<NewCareRule>): Promise<void> {
    const { error } = await supabase.from('care_rules').update(changes).eq('id', ruleId);
    if (error) throw error;
  }
}
```

```typescript
import { Component, inject, signal, computed } from '@angular/core';
import { Router, RouterLink } from '@angular/router';
import { PlantCatalogService, Plant } from './plant-catalog.service';

export function filterPlants(plants: Plant[], query: string): Plant[] {
  const q = query.trim().toLowerCase();
  if (!q) return plants;
  return plants.filter(p => p.name.toLowerCase().includes(q));
}

export function validateNewPlantForm(name: string, sunRequirements: Plant['sun_requirements']): string | null {
  if (!name.trim()) return 'Name is required.';
  if (!sunRequirements) return 'Sun requirement is required.';
  return null;
}

@Component({
  selector: 'app-plant-catalog-list',
  standalone: true,
  imports: [RouterLink],
  templateUrl: './plant-catalog-list.component.html',
})
export class PlantCatalogListComponent {
  private service = inject(PlantCatalogService);
  private router = inject(Router);
  private plants = signal<Plant[]>([]);
  query = signal('');
  visiblePlants = computed(() => filterPlants(this.plants(), this.query()));

  newPlantName = signal('');
  newPlantSun = signal<Plant['sun_requirements']>(null);
  newPlantError = signal<string | null>(null);

  constructor() {
    this.service.list().then(p => this.plants.set(p));
  }

  async cloneAndEdit(plantId: string): Promise<void> {
    const clone = await this.service.cloneToCustom(plantId);
    this.plants.update(list => [...list, clone]);
    this.router.navigate(['/plants', clone.id]);
  }

  async addCustomPlant(): Promise<void> {
    const validation = validateNewPlantForm(this.newPlantName(), this.newPlantSun());
    this.newPlantError.set(validation);
    if (validation) return;
    const plant = await this.service.addCustom({
      name: this.newPlantName(),
      species: null,
      sun_requirements: this.newPlantSun(),
      watering_frequency_days: null,
      flowering_start_month: null,
      flowering_end_month: null,
      notes: null,
    });
    this.plants.update(list => [...list, plant]);
    this.router.navigate(['/plants', plant.id]);
  }
}
```

```typescript
import { Injectable } from '@angular/core';
import { supabase } from '../core/supabase-client';

export interface Bed {
  id: string;
  user_id: string;
  name: string;
  rows: number;
  cols: number;
}

export interface Planting {
  id: string;
  user_id: string;
  bed_id: string;
  cell_row: number;
  cell_col: number;
  plant_id: string;
  planted_date: string;
  status: 'active' | 'removed';
}

export function cellKey(row: number, col: number): string {
  return `${row}:${col}`;
}

@Injectable({ providedIn: 'root' })
export class GardenService {
  async listBeds(): Promise<Bed[]> {
    const { data, error } = await supabase.from('beds').select('*').order('name');
    if (error) throw error;
    return data as Bed[];
  }

  async createBed(name: string, rows: number, cols: number): Promise<Bed> {
    const { data: userData } = await supabase.auth.getUser();
    const { data, error } = await supabase
      .from('beds')
      .insert({ name, rows, cols, user_id: userData.user!.id })
      .select()
      .single();
    if (error) throw error;
    return data as Bed;
  }

  async listActivePlantings(bedId: string): Promise<Planting[]> {
    const { data, error } = await supabase
      .from('plantings')
      .select('*')
      .eq('bed_id', bedId)
      .eq('status', 'active');
    if (error) throw error;
    return data as Planting[];
  }

  async assignPlanting(bedId: string, row: number, col: number, plantId: string, plantedDate: string): Promise<Planting> {
    const { data: userData } = await supabase.auth.getUser();
    const { data, error } = await supabase
      .from('plantings')
      .insert({
        bed_id: bedId,
        cell_row: row,
        cell_col: col,
        plant_id: plantId,
        planted_date: plantedDate,
        status: 'active',
        user_id: userData.user!.id,
      })
      .select()
      .single();
    if (error) throw error;
    return data as Planting;
  }

  async removePlanting(plantingId: string): Promise<void> {
    const { error } = await supabase.from('plantings').update({ status: 'removed' }).eq('id', plantingId);
    if (error) throw error;
  }
}
```

```typescript
import { Component, inject, signal } from '@angular/core';
```

```typescript
import { GardenService, Bed, Planting, cellKey } from './garden.service';

export function buildGrid(rows: number, cols: number, plantings: Planting[]): (Planting | null)[][] {
  const byCell = new Map(plantings.map(p => [cellKey(p.cell_row, p.cell_col), p]));
  return Array.from({ length: rows }, (_, r) =>
    Array.from({ length: cols }, (_, c) => byCell.get(cellKey(r, c)) ?? null)
  );
}

@Component({
  selector: 'app-garden-layout',
  standalone: true,
  imports: [RouterLink],
  templateUrl: './garden-layout.component.html',
})
export class GardenLayoutComponent {
  private garden = inject(GardenService);
  beds = signal<Bed[]>([]);
  gridsByBed = signal<Map<string, (Planting | null)[][]>>(new Map());

  constructor() {
    this.garden.listBeds().then(async beds => {
      this.beds.set(beds);
      const grids = new Map<string, (Planting | null)[][]>();
      for (const bed of beds) {
        const plantings = await this.garden.listActivePlantings(bed.id);
        grids.set(bed.id, buildGrid(bed.rows, bed.cols, plantings));
      }
      this.gridsByBed.set(grids);
    });
  }

  async toggleCell(bed: Bed, row: number, col: number): Promise<void> {
    const grid = this.gridsByBed().get(bed.id);
    const existing = grid?.[row]?.[col];
    if (existing) {
      await this.garden.removePlanting(existing.id);
    }
    // Assigning a new plant to an empty cell is driven from the Plant Catalog
    // screen, which passes plantId/plantedDate; this component only removes here.
  }
}
```

### Task 11: pg_cron rolling-window top-up

**As implemented:** a single `insert … select distinct` over active plantings × recurring rules × `care_rule_due_dates(rule, planted_date, current_date, task_window_end())` that skips dates already having a task of that type (`not exists`), instead of continuing from `max(due_date)` per task type. That makes it idempotent and catch-up safe, correct when two recurring rules share a task type or a one-off task of that type lies beyond the window, and it picks up recurring rules added to a custom plant after planting (tasks can't be deleted in the app, so nothing deleted is resurrected). `revoke execute … from public, anon, authenticated` so it isn't callable as an API RPC; the job command is schema-qualified. The test (rewritten) checks the cron job exists, the API roles can't execute the function, a partially-filled window is refilled exactly without duplicating a completed task, a later-added recurring rule is picked up, removed plantings get nothing, and a second run adds nothing.

```sql
create extension if not exists pg_cron;

create or replace function top_up_recurring_tasks() returns void as $$
declare
  planting plantings;
  rule care_rules;
  latest date;
begin
  for planting in select * from plantings where status = 'active' loop
    for rule in select * from care_rules where plant_id = planting.plant_id and timing_type = 'recurring' loop
      select max(due_date) into latest from tasks
        where planting_id = planting.id and task_type = rule.task_type;
      if latest is null then
        latest := planting.planted_date;
      end if;
      if latest < current_date + 90 then
        insert into tasks (id, user_id, planting_id, task_type, due_date, completed_at, google_calendar_event_id)
        select gen_random_uuid(), planting.user_id, planting.id, rule.task_type, d, null, null
        from generate_series(latest + rule.recurrence_days, current_date + 90, (rule.recurrence_days || ' days')::interval) as d;
      end if;
    end loop;
  end loop;
end;
$$ language plpgsql;
```

```typescript
import { Injectable } from '@angular/core';
import { supabase } from '../core/supabase-client';

export interface Task {
  id: string;
  user_id: string;
  planting_id: string | null;
  task_type: 'seed_start' | 'transplant' | 'prune' | 'fertilize' | 'harvest';
  due_date: string;
  completed_at: string | null;
  google_calendar_event_id: string | null;
}

export interface TaskFilter {
  taskType?: Task['task_type'];
  plantingIds?: Set<string>;
}

export function filterTasks(tasks: Task[], filter: TaskFilter): Task[] {
  return tasks.filter(t => {
    if (filter.taskType && t.task_type !== filter.taskType) return false;
    if (filter.plantingIds && (!t.planting_id || !filter.plantingIds.has(t.planting_id))) return false;
    return true;
  });
}

export function isDueWithin(dueDate: string, today: Date, days: number): boolean {
  const due = new Date(dueDate + 'T00:00:00');
  const diffDays = (due.getTime() - today.getTime()) / (1000 * 60 * 60 * 24);
  return diffDays >= 0 && diffDays <= days;
}

@Injectable({ providedIn: 'root' })
export class TaskService {
  async listAll(): Promise<Task[]> {
    const { data, error } = await supabase.from('tasks').select('*').order('due_date');
    if (error) throw error;
    return data as Task[];
  }

  async listUpcoming(days: number): Promise<Task[]> {
    const all = await this.listAll();
    return all.filter(t => isDueWithin(t.due_date, new Date(), days));
  }

  async markComplete(taskId: string): Promise<void> {
    const { error } = await supabase.from('tasks').update({ completed_at: new Date().toISOString() }).eq('id', taskId);
    if (error) throw error;
  }

  async addManualTask(taskType: Task['task_type'], dueDate: string): Promise<Task> {
    const { data: userData } = await supabase.auth.getUser();
    const { data, error } = await supabase
      .from('tasks')
      .insert({ task_type: taskType, due_date: dueDate, user_id: userData.user!.id, planting_id: null })
      .select()
      .single();
    if (error) throw error;
    return data as Task;
  }
}
```

```typescript
const TOKEN_URL = 'https://oauth2.googleapis.com/token';
const REVOKE_URL = 'https://oauth2.googleapis.com/revoke';
const CALENDAR_API = 'https://www.googleapis.com/calendar/v3';

export interface GoogleOAuthConfig {
  clientId: string;
  clientSecret: string;
}

export interface EventTask {
  due_date: string;
  task_type: string;
  completed_at?: string | null;
  google_calendar_event_id: string | null;
}

/** Google rejected the refresh token (revoked, expired, or the user removed app access). */
export class InvalidGrantError extends Error {}

/** The Samwise calendar no longer exists (the user deleted it in Google Calendar). */
export class CalendarGoneError extends Error {}

const TASK_LABELS: Record<string, string> = {
  seed_start: 'Start seeds',
  transplant: 'Transplant',
  prune: 'Prune',
  fertilize: 'Fertilize',
  harvest: 'Harvest',
};

function addOneDay(date: string): string {
  const d = new Date(`${date}T00:00:00Z`);
  d.setUTCDate(d.getUTCDate() + 1);
  return d.toISOString().slice(0, 10);
}

function expiresAt(expiresInSeconds: number): string {
  return new Date(Date.now() + expiresInSeconds * 1000).toISOString();
}

export async function exchangeCodeForTokens(code: string, config: GoogleOAuthConfig & { redirectUri: string }) {
  const response = await fetch(TOKEN_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      code,
      client_id: config.clientId,
      client_secret: config.clientSecret,
      redirect_uri: config.redirectUri,
      grant_type: 'authorization_code',
    }).toString(),
  });
  if (!response.ok) throw new Error(`code exchange failed: ${await response.text()}`);
  const json = await response.json();
  return {
    refresh_token: json.refresh_token as string,
    access_token: json.access_token as string,
    expires_at: expiresAt(json.expires_in),
  };
}

export async function refreshAccessToken(refreshToken: string, config: GoogleOAuthConfig) {
  const response = await fetch(TOKEN_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      refresh_token: refreshToken,
      client_id: config.clientId,
      client_secret: config.clientSecret,
      grant_type: 'refresh_token',
    }).toString(),
  });
  if (!response.ok) {
    const text = await response.text();
    if (text.includes('invalid_grant')) throw new InvalidGrantError(text);
    throw new Error(`refresh failed: ${text}`);
  }
  const json = await response.json();
  return { access_token: json.access_token as string, expires_at: expiresAt(json.expires_in) };
}

export async function revokeToken(token: string): Promise<void> {
  await fetch(`${REVOKE_URL}?token=${encodeURIComponent(token)}`, { method: 'POST' });
}

/** Returns the stored calendar's ID if it still exists, otherwise creates a new "Samwise" calendar. */
export async function ensureSamwiseCalendar(accessToken: string, existingCalendarId: string | null): Promise<string> {
  const headers = { Authorization: `Bearer ${accessToken}`, 'Content-Type': 'application/json' };
  if (existingCalendarId) {
    const existing = await fetch(`${CALENDAR_API}/calendars/${encodeURIComponent(existingCalendarId)}`, { headers });
    if (existing.ok) return existingCalendarId;
  }
  const created = await fetch(`${CALENDAR_API}/calendars`, {
    method: 'POST',
    headers,
    body: JSON.stringify({ summary: 'Samwise', description: 'Garden care tasks from Samwise' }),
  });
  if (!created.ok) throw new Error(`calendar create failed: ${await created.text()}`);
  return (await created.json()).id as string;
}

export async function deleteCalendar(accessToken: string, calendarId: string): Promise<void> {
  await fetch(`${CALENDAR_API}/calendars/${encodeURIComponent(calendarId)}`, {
    method: 'DELETE',
    headers: { Authorization: `Bearer ${accessToken}` },
  });
}

/** Creates or updates the task's all-day event; returns the event ID. */
export async function upsertEvent(accessToken: string, calendarId: string, task: EventTask): Promise<string> {
  const base = `${CALENDAR_API}/calendars/${encodeURIComponent(calendarId)}/events`;
  const label = TASK_LABELS[task.task_type] ?? task.task_type;
  const event = {
    summary: task.completed_at ? `✓ ${label}` : label,
    start: { date: task.due_date },
    end: { date: addOneDay(task.due_date) }, // all-day events use an exclusive end date
  };
  const response = await fetch(task.google_calendar_event_id ? `${base}/${task.google_calendar_event_id}` : base, {
    method: task.google_calendar_event_id ? 'PATCH' : 'POST',
    headers: { Authorization: `Bearer ${accessToken}`, 'Content-Type': 'application/json' },
    body: JSON.stringify(event),
  });
  if (response.status === 404 && !task.google_calendar_event_id) throw new CalendarGoneError(calendarId);
  if (response.status === 404) {
    // The event itself was deleted by the user: recreate it.
    return upsertEvent(accessToken, calendarId, { ...task, google_calendar_event_id: null });
  }
  if (!response.ok) throw new Error(`event upsert failed: ${await response.text()}`);
  return (await response.json()).id as string;
}

export async function deleteEvent(accessToken: string, calendarId: string, eventId: string): Promise<void> {
  await fetch(`${CALENDAR_API}/calendars/${encodeURIComponent(calendarId)}/events/${eventId}`, {
    method: 'DELETE',
    headers: { Authorization: `Bearer ${accessToken}` },
  });
}
```

```typescript
import { ensureSamwiseCalendar, exchangeCodeForTokens, upsertEvent } from '../_shared/google.ts';

Deno.serve(async (req) => {
  const userClient = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_ANON_KEY')!, {
    global: { headers: { Authorization: req.headers.get('Authorization') ?? '' } },
  });
  const { data: userData, error: userError } = await userClient.auth.getUser();
  if (userError || !userData.user) return new Response('Unauthorized', { status: 401 });
  const userId = userData.user.id;

  const { code } = await req.json();
  if (typeof code !== 'string' || !code) return new Response('missing code', { status: 400 });

  const tokens = await exchangeCodeForTokens(code, {
    clientId: Deno.env.get('GOOGLE_CLIENT_ID')!,
    clientSecret: Deno.env.get('GOOGLE_CLIENT_SECRET')!,
    redirectUri: Deno.env.get('GOOGLE_REDIRECT_URI')!,

  const service = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
  const { data: existing } = await service
    .from('calendar_credentials').select('calendar_id').eq('user_id', userId).maybeSingle();
  const calendarId = await ensureSamwiseCalendar(tokens.access_token, existing?.calendar_id ?? null);

  const { error: upsertError } = await service.from('calendar_credentials').upsert({
    user_id: userId,
    refresh_token: tokens.refresh_token,
    access_token: tokens.access_token,
    expires_at: tokens.expires_at,
    calendar_id: calendarId,
  });
  if (upsertError) return new Response(upsertError.message, { status: 500 });

  // Backfill: tasks created before connecting never went through the webhook with a
  // connection in place. Writing the event ID back fires the webhook, which no-ops
  // because only google_calendar_event_id changed (see Task 16's shouldSync).
  const today = new Date().toISOString().slice(0, 10);
  const { data: pending } = await service
    .from('tasks')
    .select('id, due_date, task_type, completed_at, google_calendar_event_id')
    .eq('user_id', userId)
    .is('completed_at', null)
    .is('google_calendar_event_id', null)
    .gte('due_date', today);
  for (const task of pending ?? []) {
    const eventId = await upsertEvent(tokens.access_token, calendarId, task);
    await service.from('tasks').update({ google_calendar_event_id: eventId }).eq('id', task.id);
  }

  return new Response(JSON.stringify({ connected: true, backfilled: pending?.length ?? 0 }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' },
  });
```

```typescript
import { Component, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { AuthService } from '../core/auth.service';
import { supabase } from '../core/supabase-client';
import { environment } from '../../environments/environment';

// Only lets Samwise create calendars and manage events on them — no access to the user's other calendars.
const GOOGLE_CALENDAR_SCOPE = 'https://www.googleapis.com/auth/calendar.app.created';
const OAUTH_STATE_KEY = 'samwise.calendarOAuthState';

export function parseOAuthRedirect(
  params: URLSearchParams, expectedState: string | null,
): { code: string } | { error: string } | null {
  const code = params.get('code');
  const error = params.get('error');
  if (!code && !error) return null;
  if (!expectedState || params.get('state') !== expectedState) return { error: 'state_mismatch' };
  if (error) return { error };
  return { code: code! };
}

@Component({
  selector: 'app-settings',
  standalone: true,
  templateUrl: './settings.component.html',
})
export class SettingsComponent {
  auth = inject(AuthService);
  private router = inject(Router);
  connecting = signal(false);
  connectError = signal<string | null>(null);

  constructor() {
    const redirect = parseOAuthRedirect(
      new URLSearchParams(window.location.search),
      sessionStorage.getItem(OAUTH_STATE_KEY),
    );
    if (redirect) {
      sessionStorage.removeItem(OAUTH_STATE_KEY);
      // Drop ?code=...&state=... from the address bar.
      this.router.navigate([], { queryParams: {}, replaceUrl: true });
      if ('code' in redirect) this.finishConnect(redirect.code);
      else this.connectError.set(redirect.error);
    }
  }

  connectCalendar(): void {
    this.connecting.set(true);
    const state = crypto.randomUUID();
    sessionStorage.setItem(OAUTH_STATE_KEY, state);
    const params = new URLSearchParams({
      client_id: environment.googleClientId,
      redirect_uri: `${window.location.origin}/settings`,
      response_type: 'code',
      access_type: 'offline',
      prompt: 'consent', // always return a refresh token, even on reconnect
      scope: GOOGLE_CALENDAR_SCOPE,
      state,
    });
    window.location.href = `https://accounts.google.com/o/oauth2/v2/auth?${params.toString()}`;
  }

  protected async finishConnect(code: string): Promise<void> {
    this.connecting.set(true);
    const { data: sessionData } = await supabase.auth.getSession();
    const response = await fetch(`${environment.supabaseUrl}/functions/v1/connect-calendar`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${sessionData.session?.access_token}`,
      },
      body: JSON.stringify({ code }),
    });
    if (!response.ok) this.connectError.set('connect_failed');
    this.connecting.set(false);
  }
}
```

```typescript
import {
  CalendarGoneError, EventTask, GoogleOAuthConfig, InvalidGrantError, refreshAccessToken, upsertEvent,
} from '../_shared/google.ts';

export interface SyncCredentials {
  refresh_token: string;
  calendar_id: string;
}

export type SyncResult = { status: 'ok'; eventId: string } | { status: 'disconnected' } | { status: 'failed' };

export function isAuthorizedWebhook(header: string | null, secret: string | undefined): boolean {
  return !!secret && header === secret;
}

export function shouldSync(oldRecord: Partial<EventTask> | null, newRecord: EventTask): boolean {
  if (!oldRecord) return true;
  return oldRecord.due_date !== newRecord.due_date
    || oldRecord.task_type !== newRecord.task_type
    || (oldRecord.completed_at ?? null) !== (newRecord.completed_at ?? null);
}

export async function syncTaskToCalendar(
  task: EventTask, credentials: SyncCredentials, config: GoogleOAuthConfig,
): Promise<SyncResult> {
  try {
    const { access_token } = await refreshAccessToken(credentials.refresh_token, config);
    return { status: 'ok', eventId: await upsertEvent(access_token, credentials.calendar_id, task) };
  } catch (e) {
    if (e instanceof InvalidGrantError || e instanceof CalendarGoneError) return { status: 'disconnected' };
    return { status: 'failed' };
  }
}
```

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';
import { deleteEvent, GoogleOAuthConfig, refreshAccessToken } from '../_shared/google.ts';
import { isAuthorizedWebhook, shouldSync, syncTaskToCalendar } from './sync.ts';

Deno.serve(async (req) => {
  // JWT verification is off for this function (config.toml); the shared secret is the auth.
  if (!isAuthorizedWebhook(req.headers.get('x-webhook-secret'), Deno.env.get('CALENDAR_SYNC_WEBHOOK_SECRET'))) {
    return new Response('Unauthorized', { status: 401 });
  }

  const { type, table, record, old_record } = await req.json();
  if (table !== 'tasks') return new Response('ignored', { status: 200 });

  // record is null on DELETE, old_record is null on INSERT.
  const userId: string = (record ?? old_record).user_id;
  const service = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
  const config: GoogleOAuthConfig = {
    clientId: Deno.env.get('GOOGLE_CLIENT_ID')!,
    clientSecret: Deno.env.get('GOOGLE_CLIENT_SECRET')!,
  };

  const { data: credentials } = await service
    .from('calendar_credentials').select('refresh_token, calendar_id').eq('user_id', userId).maybeSingle();
  if (!credentials) return new Response('no calendar connected', { status: 200 });

  if (type === 'DELETE') {
    if (old_record.google_calendar_event_id) {
      try {
        const { access_token } = await refreshAccessToken(credentials.refresh_token, config);
        await deleteEvent(access_token, credentials.calendar_id, old_record.google_calendar_event_id);
      } catch { /* best effort */ }
    }
    return new Response('ok', { status: 200 });
  }

  if (!shouldSync(old_record, record)) return new Response('no-op', { status: 200 });

  const result = await syncTaskToCalendar(record, credentials, config);
  if (result.status === 'disconnected') {
    // Token revoked or Samwise calendar deleted: drop the connection so Settings shows "reconnect".
    await service.from('calendar_credentials').delete().eq('user_id', userId);
  } else if (result.status === 'ok' && result.eventId !== record.google_calendar_event_id) {
    await service.from('tasks').update({ google_calendar_event_id: result.eventId }).eq('id', record.id);
  }
  return new Response(result.status, { status: 200 });
});
```

```sql
create extension if not exists pg_net;

-- security definer: the trigger fires as the signed-in user, who can't read Vault.
create or replace function notify_calendar_sync() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  sync_url text;
  webhook_secret text;
begin
  select decrypted_secret into sync_url
    from vault.decrypted_secrets where name = 'calendar_sync_url';
  select decrypted_secret into webhook_secret
    from vault.decrypted_secrets where name = 'calendar_sync_webhook_secret';

  -- Not configured in this environment (e.g. fresh local reset): skip silently.
  if sync_url is null or webhook_secret is null then
    return coalesce(new, old);
  end if;

  perform net.http_post(
    url := sync_url,
    headers := jsonb_build_object(
      'Content-Type', 'application/json',
      'x-webhook-secret', webhook_secret
    ),
    body := jsonb_build_object(
      'type', tg_op,
      'table', tg_table_name,
      'record', case when tg_op = 'DELETE' then null else to_jsonb(new) end,
      'old_record', case when tg_op = 'INSERT' then null else to_jsonb(old) end
    )
  );
  return coalesce(new, old);
end;
$$;

create trigger tasks_after_change_notify_calendar
  after insert or update or delete on tasks
  for each row execute function notify_calendar_sync();
```

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';
import { deleteCalendar, refreshAccessToken, revokeToken } from '../_shared/google.ts';

Deno.serve(async (req) => {
  const userClient = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_ANON_KEY')!, {
    global: { headers: { Authorization: req.headers.get('Authorization') ?? '' } },
  });
  const { data: userData, error: userError } = await userClient.auth.getUser();
  if (userError || !userData.user) return new Response('Unauthorized', { status: 401 });
  const userId = userData.user.id;

  const service = createClient(Deno.env.get('SUPABASE_URL')!, Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!);
  const { data: credentials } = await service
    .from('calendar_credentials').select('refresh_token, calendar_id').eq('user_id', userId).maybeSingle();

  if (credentials) {
    // Delete the row first so the webhook ignores the event-ID cleanup below.
    await service.from('calendar_credentials').delete().eq('user_id', userId);
    try {
      const { access_token } = await refreshAccessToken(credentials.refresh_token, {
        clientId: Deno.env.get('GOOGLE_CLIENT_ID')!,
        clientSecret: Deno.env.get('GOOGLE_CLIENT_SECRET')!,
      });
      await deleteCalendar(access_token, credentials.calendar_id);
    } catch { /* token already invalid: nothing left to delete with it */ }
    await revokeToken(credentials.refresh_token).catch(() => null);
  }

  // So a later reconnect backfills every upcoming task into the new calendar.
  await service.from('tasks').update({ google_calendar_event_id: null })
    .eq('user_id', userId).not('google_calendar_event_id', 'is', null);

  return new Response(JSON.stringify({ connected: false }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' },
  });
});
```
